{\normalsize\bfseries Formulário resumido -- Estatística Não-Paramétrica}\par
\smallskip
\textbf{Convenções.} $X_1,\dots,X_n$ i.i.d. com função de distribuição (FD) $F$. Toda função aparece com o seu argumento: $F(y)$, $g(y)$, $K(u)$, etc. A letra $y$ é sempre a variável de integração ou o argumento de uma FD; $x_0$ é um ponto fixo.

\textbf{Ordens.} $b_n>0$ é uma sequência determinística.
\begin{itemize}
  \item \emph{Números} (viés, variância, MSE, $\E[\cdot]$, $\P(\cdot)$), com $a_n\ge0$:
  $a_n=O(b_n)$ se existe $C>0$, que não depende de $n$, com $a_n\le C\,b_n$ para $n$ grande; \ $a_n=o(b_n)$ se $a_n/b_n\to0$.
  O mesmo vale com $h\to0$ no lugar de $n\to\infty$, e com $C$ que não depende de $x$ (cota uniforme).
  \item \emph{V.a., trajetória a trajetória:} $Z_n=O(b_n)$ q.c. se, para quase todo $\omega$, a sequência de números $|Z_n(\omega)|$ é $O(b_n)$.
  \item \emph{V.a., em probabilidade:} $Z_n=O_P(b_n)$ se, para todo $\eta>0$, existem $M,N$ com $\P(|Z_n|>M\,b_n)<\eta$ para todo $n\ge N$; \ $Z_n=o_P(b_n)$ se $Z_n/b_n\pto0$.
  \item \emph{Relações:} $Z_n=O(b_n)$ q.c. $\Rightarrow Z_n=O_P(b_n)$. Se $b_n\to0$: $O(b_n)$ q.c. $\Rightarrow Z_n\qc0$, e $O_P(b_n)\Rightarrow Z_n\pto0$ (mas não $\qc$).
\end{itemize}

\section{Convergência}\secrule
\begin{description}
  \item[Em probabilidade] $X_n\pto X$ se, para todo $\varepsilon>0$, \ $\P(|X_n-X|>\varepsilon)\to0$ quando $n\to\infty$.
  \item[Quase certa] $X_n\qc X$ se \ $\P\big(\lim_{n\to\infty}|X_n-X|=0\big)=1$.
  \item[Quase certa (eventos de cauda)] Equivalentemente, para todo $\varepsilon>0$,
  \[\P\big(|X_n-X|>\varepsilon\ \text{infinitas vezes}\big)=0,\]
  em que $\{A_n\text{ i.v.}\}=\limsup_nA_n=\bigcap_{m\ge1}\bigcup_{n\ge m}A_n$. Também equivale a $\P\big(\sup_{n\ge m}|X_n-X|>\varepsilon\big)\to0$ quando $m\to\infty$.
  \item[Em distribuição] $X_n\dto X$ se $\P(X_n\le t)\to\P(X\le t)$ em todo ponto $t$ de continuidade de $t\mapsto\P(X\le t)$.
  \item[Relações] $X_n\qc X\Rightarrow X_n\pto X\Rightarrow X_n\dto X$; \ e $X_n\dto c$ (constante) $\Rightarrow X_n\pto c$.
\end{description}

\section{Ferramentas}\secrule
\begin{description}
  \item[Markov] Denotando $\nu_{nr}=\E|T_n-\theta|^r<\infty$, para algum $r>0$, vale, para $t>0$,
  \[\P\big(|T_n-\theta|>t\big)\le\frac{\nu_{nr}}{t^r}.\]
  \item[Chebyshev] Caso $r=2$ com $\E[T_n]=\theta$: \quad $\P(|T_n-\theta|>t)\le \dfrac{\V(T_n)}{t^2}$.
  \item[Hoeffding] $Y_1,\dots,Y_n$ independentes, $\E[Y_i]=0$ e $a_i\le Y_i\le b_i$. Para $\varepsilon>0$ e qualquer $t>0$,
  \[\P\Big(\sum_{i=1}^nY_i\ge\varepsilon\Big)\le e^{-t\varepsilon}\prod_{i=1}^ne^{t^2(b_i-a_i)^2/8}.\]
  Escolhendo $t=4\varepsilon/\sum_i(b_i-a_i)^2$: \ $\le\exp\!\big(-2\varepsilon^2/\sum_i(b_i-a_i)^2\big)$. \\
  Caso das aulas: $Y_i\in[0,1]$, aplicado a $Y_i-\E[Y_i]$ e a $-(Y_i-\E[Y_i])$ com $\varepsilon$ trocado por $n\varepsilon$:
  \ $\P\big(|\bar Y-\E[\bar Y]|\ge\varepsilon\big)\le2e^{-2n\varepsilon^2}$.
  \item[Borel-Cantelli] Se $\sum_n\P(A_n)<\infty$, então $\P(A_n\text{ i.v.})=0$. \\
  \emph{Uso:} com $A_n=\{|T_n-\theta|>\varepsilon\}$ e a forma de cauda acima: se $\sum_n\P(|T_n-\theta|>\varepsilon)<\infty$ para todo $\varepsilon>0$, então $T_n\qc\theta$. Toda cota do tipo $Ce^{-cn\varepsilon^2}$ é somável em $n$.
  \item[Método delta] Se $\sqrt n(Y_n-\mu)\dto N(0,\sigma^2)$ e $g$ é derivável em $\mu$:
  \[\sqrt n\big(g(Y_n)-g(\mu)\big)\dto N\big(0,\ g'(\mu)^2\sigma^2\big).\]
  \item[Viés e variância] $\MSE(\hat\theta)=\E[(\hat\theta-\theta)^2]=\big(\E[\hat\theta]-\theta\big)^2+\V(\hat\theta)$.
\end{description}

\section{Distribuição empírica}\secrule
\begin{description}
  \item[Definição] $\displaystyle \Fn(y)=\frac1n\sum_{i=1}^n\I(X_i\le y)$, \ para cada $y\in\R$.
  \item[Num ponto $y$ fixo] $n\Fn(y)\sim\textrm{Binomial}\big(n,F(y)\big)$, logo
  \[\E[\Fn(y)]=F(y),\qquad \V[\Fn(y)]=\frac{F(y)\big(1-F(y)\big)}{n}\le\frac1{4n}.\]
  \item[Erro máximo] $D_n=\sup_{y\in\R}|\Fn(y)-F(y)|$.
  \item[Glivenko-Cantelli] $D_n\qc0$.
  \item[DKW] $\P(D_n\ge\varepsilon)\le2e^{-2n\varepsilon^2}$, para todo $\varepsilon>0$ e todo $n$.
  \item[Taxa q.c.] Quase certamente, $D_n\le2\sqrt{\log n/n}$ para todo $n$ suficientemente grande. Isto é, $D_n=O\big(\sqrt{\log n/n}\big)$ q.c.
  \item[Taxa em probabilidade] Por DKW, $\P(\sqrt n\,D_n\ge M)\le2e^{-2M^2}$ para todo $n$, logo $D_n=O_P(n^{-1/2})$: taxa melhor, modo mais fraco que a q.c.
\end{description}

\subsection{Molde: consistência por Lipschitz}
Se existe $L>0$ tal que, para quaisquer FDs $G_1,G_2$,
\[|T(G_1)-T(G_2)|\le L\sup_{y}|G_1(y)-G_2(y)|,\]
então $|T(\Fn)-T(F)|\le L\,D_n$, logo $T(\Fn)\qc T(F)$, com $|T(\Fn)-T(F)|=O\big(\sqrt{\log n/n}\big)$ q.c. e $|T(\Fn)-T(F)|=O_P(n^{-1/2})$.

\section{Dimensão VC}\secrule
$\mathcal C$ é uma classe de conjuntos e $A=\{x_1,\dots,x_k\}$ é um conjunto finito de pontos.
\begin{description}
  \item[Fragmentar] $\mathcal C$ fragmenta $A$ se \emph{todo} subconjunto de $A$ é da forma $C\cap A$ para algum $C\in\mathcal C$.
  \item[Coeficiente] $s(\mathcal C,n)=$ número máximo de subconjuntos distintos $C\cap A$, entre conjuntos $A$ com $n$ pontos. Sempre $s(\mathcal C,n)\le2^n$.
  \item[Dimensão VC] $\VC(\mathcal C)=\sup\{n:s(\mathcal C,n)=2^n\}$, isto é, o maior $n$ tal que \emph{algum} conjunto de $n$ pontos é fragmentado.
  \item[Sauer-Shelah] Se $\VC(\mathcal C)=d<\infty$, então $s(\mathcal C,n)\le(n+1)^d$.
  \item[Desigualdade de VC] Com $P_n(C)=\frac1n\sum_i\I(X_i\in C)$:
  \[\P\Big(\sup_{C\in\mathcal C}|P_n(C)-P(C)|>\varepsilon\Big)\le8\,s(\mathcal C,n)\,e^{-n\varepsilon^2/32}.\]
  Com Sauer-Shelah e Borel-Cantelli: $\VC(\mathcal C)<\infty\Rightarrow\sup_{C\in\mathcal C}|P_n(C)-P(C)|\qc0$, para toda $P$.
\end{description}
\begin{center}
\begin{tabular}{lc}
\toprule
Classe $\mathcal C$ & $\VC(\mathcal C)$ \\
\midrule
semirretas $(-\infty,a]$ em $\R$ & $1$ \\
intervalos $[a,b]$ em $\R$ & $2$ \\
retângulos $[a_1,b_1]\times[a_2,b_2]$ & $4$ \\
semiespaços em $\R^d$ & $d+1$ \\
\bottomrule
\end{tabular}
\end{center}
\textbf{Receita para mostrar $\VC(\mathcal C)=d$:} (i) exiba \emph{um} conjunto de $d$ pontos e obtenha todos os seus subconjuntos; (ii) para \emph{qualquer} conjunto de $d+1$ pontos, ordene $x_1<\dots<x_{d+1}$ e encontre um subconjunto que nenhum $C\in\mathcal C$ produz.

\section{Funcionais e função de influência}\secrule
\begin{description}
  \item[Funcional] Uma função $T$ que recebe uma FD e devolve um número: $\theta=T(F)$. O estimador plug-in é $T(\Fn)$.
  \item[Funcional linear] Existe uma função $g:\R\to\R$ tal que
  \[T(F)=\int g(y)\,dF(y)=\E_F[g(X_1)].\]
  O plug-in é a média amostral de $g$: \ $T(\Fn)=\frac1n\sum_{i=1}^ng(X_i)$. \\
  \emph{Exemplos:} média, $g(y)=y$; \ $F(t)$ com $t$ fixo, $g(y)=\I(y\le t)$.
  \item[Massa pontual em $x_0$] É a FD de uma v.a. $Z$ que vale sempre $x_0$:
  \[\delta_{x_0}(y)=\P(Z\le y)=\I(x_0\le y),\qquad y\in\R.\]
  Vale $0$ para $y<x_0$ e $1$ para $y\ge x_0$. É o $\Fn$ de uma única observação em $x_0$.
  \item[Contaminação] Para $\varepsilon\in[0,1]$, a FD
  \[F_{\varepsilon}(y)=(1-\varepsilon)F(y)+\varepsilon\,\delta_{x_0}(y),\qquad y\in\R,\]
  sorteia de $F$ com prob.\ $1-\varepsilon$ e devolve $x_0$ com prob.\ $\varepsilon$.
  \item[Função de influência] Efeito em $T$ de contaminar $F$ no ponto $x_0$:
  \[\IF(x_0)=\lim_{\varepsilon\downarrow0}\frac{T(F_\varepsilon)-T(F)}{\varepsilon}=\frac{d}{d\varepsilon}T(F_\varepsilon)\Big|_{\varepsilon=0}.\]
  Sempre vale $\int\IF(y)\,dF(y)=0$: use para conferir a conta.
\end{description}

\begin{center}
\begin{tabular}{ll}
\toprule
$T(F)$ & $\IF(x_0)$ \\
\midrule
$\int g(y)\,dF(y)$ & $g(x_0)-T(F)$ \\
média $\mu$ & $x_0-\mu$ \\
variância $\sigma^2$ & $(x_0-\mu)^2-\sigma^2$ \\
\bottomrule
\end{tabular}
\end{center}

\begin{description}
  \item[Regra da cadeia] Se $T(F)=h(\theta_1,\dots,\theta_k)$ com $\theta_j=\int g_j(y)\,dF(y)$ e $h$ diferenciável:
  \[\IF(x_0)=\sum_{j=1}^k\frac{\partial h}{\partial\theta_j}(\theta_1,\dots,\theta_k)\,\big(g_j(x_0)-\theta_j\big).\]
  \item[Expansão de von Mises] Suposição: $T(G)-T(F)=\int\IF(y)\,d(G-F)(y)+R(G,F)$, com $\sqrt n\,R(\Fn,F)=o_P(1)$ e $\tau^2=\int\IF(y)^2dF(y)\in(0,\infty)$.
  \item[Normalidade assintótica] Sob essa suposição,
  \[\sqrt n\big(T(\Fn)-T(F)\big)=\frac1{\sqrt n}\sum_{i=1}^n\IF(X_i)+o_P(1)\dto N(0,\tau^2).\]
  Em particular, $T(\Fn)-T(F)=O_P(n^{-1/2})$ (convergência em distribuição implica $O_P(1)$).
  \item[Intervalo de confiança] Se $\hat\tau^2=\frac1n\sum_{i=1}^n\widehat{\IF}(X_i)^2$ é consistente para $\tau^2$ ($\widehat{\IF}$: IF calculada em $\Fn$):
  \ $T(\Fn)\pm z_{1-\alpha/2}\,\hat\tau/\sqrt n$.
\end{description}

\section{Jackknife e bootstrap}\secrule
$\hat\theta_n=T(\Fn)$; \ $\hat\theta_{-i}$ é o mesmo estimador calculado \emph{sem} $X_i$; \ $\bar\theta=\frac1n\sum_{i=1}^n\hat\theta_{-i}$.
\begin{description}
  \item[Jackknife] Viés: \ $\hat b_{\textrm{jack}}=(n-1)(\bar\theta-\hat\theta_n)$. \\
  Estimador corrigido: \ $\hat\theta_{\textrm{jack}}=\hat\theta_n-\hat b_{\textrm{jack}}=n\hat\theta_n-(n-1)\bar\theta$. \\
  Variância: \ $\hat v_{\textrm{jack}}=\frac{n-1}{n}\sum_{i=1}^n(\hat\theta_{-i}-\bar\theta)^2$.
  \item[Por que funciona] Se $\E[\hat\theta_n]=\theta+\frac an+\frac b{n^2}+O(n^{-3})$, com $a,b$ que não dependem de $n$, então $\E[\hat b_{\textrm{jack}}]=\frac an+O(n^{-2})$ e $\E[\hat\theta_{\textrm{jack}}]-\theta=O(n^{-2})$, enquanto o viés de $\hat\theta_n$ é $\frac an+O(n^{-2})$.
  \item[Identidade útil] $\bar X_{-i}=\dfrac{n\bar X-X_i}{n-1}$.
  \item[Bootstrap] Sorteie $X_1^*,\dots,X_n^*$ com reposição de $X_1,\dots,X_n$ (i.i.d. de $\Fn$) e calcule $\hat\theta^*=T(\Fn^*)$. Repita $B$ vezes: $\hat\theta^*_1,\dots,\hat\theta^*_B$; \ $\bar\theta^*=\frac1B\sum_b\hat\theta^*_b$.
  \item[Erro padrão] $\hat{\textrm{se}}_{\textrm{boot}}^2=\frac1{B-1}\sum_{b=1}^B\big(\hat\theta^*_b-\bar\theta^*\big)^2$.
  \item[Viés] $\hat b^*=\E^*[\hat\theta^*]-\hat\theta_n$ ($\E^*$: esperança sob $\Fn$), aproximado por $\bar\theta^*-\hat\theta_n$.
  \item[Intervalos] t: $\hat\theta_n\pm z_{1-\alpha/2}\,\hat{\textrm{se}}_{\textrm{boot}}$; \ percentil: $[q^*_{\alpha/2},\,q^*_{1-\alpha/2}]$, quantis amostrais dos $\hat\theta^*_b$.
\end{description}

\section{Estimador de densidade por kernel}\secrule
\begin{description}
  \item[Kernel] $K:\R\to\R$ integrável com $\int K(u)\,du=1$ e $\int K(u)^2du<\infty$. Notação: $\kappa_j=\int|u|^j|K(u)|\,du$.
  \item[Ordem $k$] $\kappa_k<\infty$ e $\int u^jK(u)\,du=0$ para $j=1,\dots,k$. Simétrico com $\kappa_1<\infty$ $\Rightarrow$ ordem $1$.
  \item[KDE] Com banda $h>0$:
  \[\hat f_n(x)=\frac1{nh}\sum_{i=1}^nK\Big(\frac{X_i-x}{h}\Big).\]
  \item[Esperança e variância] Trocando $u=(y-x)/h$:
  \[\E[\hat f_n(x)]=\int K(u)\,f(x+uh)\,du,\]
  \[\V[\hat f_n(x)]\le\frac1{nh}\int K(u)^2f(x+uh)\,du\le\frac{\sup_yf(y)\int K(u)^2du}{nh}.\]
  \item[Hölder $\Sigma(\beta,L)$] $\ell$ é o maior inteiro \emph{estritamente} menor que $\beta$; $f$ é $\ell$ vezes derivável e $|f^{(\ell)}(y)-f^{(\ell)}(x)|\le L|y-x|^{\beta-\ell}$. Para $\beta\le1$: $\ell=0$ e $|f(y)-f(x)|\le L|y-x|^\beta$.
  \item[Hipótese] $f\in\Sigma(\beta,L)$ e $K$ limitado, de ordem $\ell$, com $\kappa_\beta<\infty$. ($C$ depende só de $\beta,L,K$.)
  \item[Viés] $\big|\E[\hat f_n(x)]-f(x)\big|\le C\,h^{\beta}$ \ (com $C=L\kappa_\beta/\ell!$).
  \item[Variância] $f$ é limitada, logo $\V[\hat f_n(x)]\le\dfrac{C}{nh}$.
  \item[Risco] $\MSE(\hat f_n(x))\le C\big(h^{2\beta}+\frac1{nh}\big)$ para todos $x,h$. Com $h=n^{-1/(2\beta+1)}$ os dois termos são iguais e
  \[\sup_x\MSE(\hat f_n(x))\le2C\,n^{-\frac{2\beta}{2\beta+1}}=O\big(n^{-\frac{2\beta}{2\beta+1}}\big).\]
  \item[Em probabilidade] Markov com $r=2$ ($\nu_{n2}=\MSE$): $\P\big(|\hat f_n(x)-f(x)|>M\,n^{-\frac{\beta}{2\beta+1}}\big)\le\frac{2C}{M^2}$, logo $|\hat f_n(x)-f(x)|=O_P\big(n^{-\frac{\beta}{2\beta+1}}\big)$ e $\hat f_n(x)\pto f(x)$.
  \item[Validação cruzada] $\hat f_{-i}(x)=\frac1{(n-1)h}\sum_{j\ne i}K(\frac{X_j-x}h)$ (sem $X_i$):
  \[\CV(h)=\int\hat f_n(x)^2\,dx-\frac2n\sum_{i=1}^n\hat f_{-i}(X_i),\]
  \[\E[\CV(h)]=\E\!\int\big(\hat f_n(x)-f(x)\big)^2dx-\int f(x)^2dx.\]
\end{description}

\subsection{Roteiro para viés/variância/MSE}
\begin{enumerate}[leftmargin=10pt,itemsep=0pt,topsep=1pt]
  \item Escrever a esperança como integral em $y$.
  \item Trocar $u=(y-x)/h$, \ $dy=h\,du$.
  \item Usar $\int K(u)\,du=1$ para escrever $\E[\hat f_n(x)]-f(x)=\int K(u)\big(f(x+uh)-f(x)\big)du$.
  \item Cotar $|f(x+uh)-f(x)|$ pela suavidade.
  \item Variância: $\V(Z)\le\E[Z^2]$.
  \item Escolher $h$ que iguala viés$^2$ e variância.
\end{enumerate}

\section{Regressão: linear smoothers}\secrule
Modelo: $Y_i=r(X_i)+\varepsilon_i$, \ $\E[\varepsilon_i\mid X_i]=0$, \ $\V[\varepsilon_i\mid X_i]=\sigma^2$; dado $\X=(X_1,\dots,X_n)$, os $\varepsilon_i$ são independentes.
\begin{description}
  \item[Linear smoother] $\hat r_n(x)=\sum_{i=1}^n\ell_i(x)\,Y_i$, com pesos $\ell_i(x)$ que dependem de $x$ e dos $X_i$, mas não dos $Y_i$.
  \item[Nadaraya-Watson] $\ell_i(x)=\dfrac{K\big((X_i-x)/h\big)}{\sum_{s=1}^nK\big((X_s-x)/h\big)}$.
  \item[Viés e variância (dado $\X$)] Se $\sum_i\ell_i(x)=1$:
  \[\E[\hat r_n(x)\mid\X]-r(x)=\sum_{i=1}^n\ell_i(x)\big(r(X_i)-r(x)\big),\]
  \[\V[\hat r_n(x)\mid\X]=\sigma^2\sum_{i=1}^n\ell_i(x)^2.\]
  \item[Risco ($\beta\le1$)] Se $r\in\Sigma(\beta,L)$ e, para $x$ e $h$ fixos: $\sum_i\ell_i(x)=1$; $\ell_i(x)=0$ quando $|X_i-x|>h$; $\sum_i|\ell_i(x)|\le C_0$; $\max_i|\ell_i(x)|\le\frac{C_0}{nh}$. Então
  \[|\textrm{viés}|\le LC_0h^\beta,\qquad \V[\hat r_n(x)\mid\X]\le\frac{\sigma^2C_0^2}{nh}.\]
  Mesma taxa do KDE. (Para $\beta>1$, exigir também $\sum_i\ell_i(x)(X_i-x)^j=0$, $j=1,\dots,\ell$.)
  \item[Validação cruzada] Se $\ell_j(x)=g_j(x)/\sum_sg_s(x)$, com $g_j(x)\ge0$ dependendo só de $x$ e $X_j$ (ex.: Nadaraya-Watson), ou para mínimos quadrados; com $L_{ii}=\ell_i(X_i)$:
  \[\CV=\frac1n\sum_{i=1}^n\Big(\frac{Y_i-\hat r_n(X_i)}{1-L_{ii}}\Big)^2.\]
\end{description}
